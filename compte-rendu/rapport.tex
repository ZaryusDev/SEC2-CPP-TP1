\section{Introduction}

% Contexte du TP, objectifs, organisation du rapport.
Ce TP a pour objectif de construire le lexique (ensemble des mots distincts et
leur nombre d'occurrences) d'un texte littéraire, à partir des fichiers
\texttt{lesMiserables\_A.txt} et \texttt{notreDameDeParis\_A.txt}.

\begin{cadregris}
  \textbf{Objectifs :}
  \begin{itemize}
    \item lire un fichier texte en C++ ;
    \item normaliser les mots (minuscules, suppression de la ponctuation) ;
    \item construire et interroger un lexique ;
    \item organiser le code en compilation séparée avec un \texttt{Makefile}.
  \end{itemize}
\end{cadregris}
